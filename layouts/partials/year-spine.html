{{- /* The year spine. The site's problem on day one is that it looks new when it is not. A list of recent posts cannot say "this has been going on since 2010"; a register of the years, each as tall as the writing done in it, can. It also has to be honest. There is a real gap between 2014 and 2019 when nothing was published, and a chart that quietly skipped those years would be a nicer shape and a false one. Empty years are drawn empty. Pure CSS — the bar height is handed over as a custom property. No script. Takes every dated page, posts and archive together, because the point is the whole decade rather than either half of it. Undated pages are excluded explicitly: a page with no date reports year 1, and letting one through makes this loop draw two thousand columns. */ -}} {{- $byYear := dict -}} {{- range where site.RegularPages "Kind" "page" -}} {{- if gt .Date.Year 1970 -}} {{- $y := .Date.Format "2006" -}} {{- $byYear = merge $byYear (dict $y (add (int (index $byYear $y | default 0)) 1)) -}} {{- end -}} {{- end -}} {{- /* The Kereső Világ entries count too. They are the reason 2014-2018 is not empty: the writing carried on through those years, it just carried on somewhere the author does not own. A spine that showed a gap there would be drawing a break that never happened. */ -}} {{- with hugo.Data.kereses -}} {{- range .irasok -}} {{- $y := substr .datum 0 4 -}} {{- $byYear = merge $byYear (dict $y (add (int (index $byYear $y | default 0)) 1)) -}} {{- end -}} {{- end -}} {{- if $byYear -}} {{- $years := slice -}} {{- range $y, $n := $byYear }}{{ $years = $years | append (int $y) }}{{ end -}} {{- $sorted := sort $years -}} {{- $from := index $sorted 0 -}} {{- $to := index $sorted (sub (len $sorted) 1) -}} {{- $now := int (now.Format "2006") -}} {{- if gt $now $to }}{{ $to = $now }}{{ end -}} {{- $max := 1 -}} {{- range $y, $n := $byYear }}{{ if gt $n $max }}{{ $max = $n }}{{ end }}{{ end -}}
    {{- range seq $from $to }} {{- $y := printf "%d" . -}} {{- $n := int (index $byYear $y | default 0) -}}
  1. {{- if gt $n 0 -}} {{ $n }} {{ $y }}{{ printf "%02d" (mod . 100) }} {{- else -}} {{ $y }}{{ printf "%02d" (mod . 100) }} {{- end -}}
    2. {{- end }}
{{- end -}}
